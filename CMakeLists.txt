cmake_minimum_required(VERSION 3.25)
project(Chtholly LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC)
  message(FATAL_ERROR "Bootstrap requires Windows x64 and the MSVC x64 toolchain")
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  message(FATAL_ERROR "Bootstrap's supported host compiler is MSVC")
endif()
if(NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "Enter an MSVC x64 Developer Shell; ARM64 and x86 hosts are unsupported")
endif()
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
  message(FATAL_ERROR "Bootstrap supports the single-configuration Ninja generator")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
  message(FATAL_ERROR "Choose CMAKE_BUILD_TYPE=Debug or Release")
endif()

include(cmake/Toolchain.cmake)
include(cmake/Targets.cmake)

add_subdirectory(toolchain/source)
add_subdirectory(toolchain/diagnostic)
add_subdirectory(toolchain/lex)
add_subdirectory(toolchain/parse)
add_subdirectory(toolchain/sem_ir)
add_subdirectory(toolchain/check)
add_subdirectory(toolchain/codegen)
add_subdirectory(toolchain/driver)
