cmake_minimum_required(VERSION 3.25)
project(Chtholly LANGUAGES C CXX)

option(CHTHOLLY_FRONTEND_ONLY "Build portable frontend verification libraries only" OFF)
option(CHTHOLLY_STATIC_ANALYSIS "Allow the pinned Windows clang-cl analysis configuration" OFF)
set(CHTHOLLY_SANITIZER "none" CACHE STRING "none, address, or undefined")
set_property(CACHE CHTHOLLY_SANITIZER PROPERTY STRINGS none address undefined)
include(CTest)
if(NOT CMAKE_GENERATOR STREQUAL "Ninja")
  message(FATAL_ERROR "Use the single-configuration Ninja generator")
endif()
if(CHTHOLLY_FRONTEND_ONLY)
  if(CHTHOLLY_STATIC_ANALYSIS OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
     NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Frontend verification requires Linux Clang; this is not a Linux product build")
  endif()
elseif(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The product requires Windows x64 with the MSVC ABI")
elseif(CHTHOLLY_STATIC_ANALYSIS)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR
     NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "18.1.8" OR
     NOT CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC" OR
     NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "Analysis requires clang-cl 18.1.8")
  endif()
elseif(NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" OR
       NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
  message(FATAL_ERROR "Enter an MSVC x64 Developer Shell")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release|RelWithDebInfo)$")
  message(FATAL_ERROR "Choose Debug, Release, or sanitizer RelWithDebInfo")
endif()
if(CHTHOLLY_SANITIZER STREQUAL "address")
  if(CHTHOLLY_FRONTEND_ONLY OR CHTHOLLY_STATIC_ANALYSIS OR
     NOT CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    message(FATAL_ERROR "AddressSanitizer requires MSVC RelWithDebInfo")
  endif()
elseif(CHTHOLLY_SANITIZER STREQUAL "undefined")
  if(NOT CHTHOLLY_FRONTEND_ONLY)
    message(FATAL_ERROR "UBSan is supported only for Linux frontend verification")
  endif()
elseif(NOT CHTHOLLY_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown CHTHOLLY_SANITIZER=${CHTHOLLY_SANITIZER}")
endif()
if(CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo" AND CHTHOLLY_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "RelWithDebInfo is reserved for sanitizer verification")
endif()

include(cmake/LLVMSource.cmake)
if(NOT CHTHOLLY_FRONTEND_ONLY)
  include(cmake/Toolchain.cmake)
else()
  find_package(LLVM 18.1.8 EXACT CONFIG REQUIRED)
  get_filename_component(locked_llvm_headers "${CHTHOLLY_LLVM_SOURCE_DIR}/llvm/include" REALPATH)
  get_filename_component(configured_llvm_headers "${LLVM_MAIN_INCLUDE_DIR}" REALPATH)
  if(NOT configured_llvm_headers STREQUAL locked_llvm_headers)
    message(FATAL_ERROR "Frontend LLVM must be built from the selected locked checkout")
  endif()
  separate_arguments(CHTHOLLY_LLVM_DEFINITIONS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
  message(STATUS "Numeric LLVM: ${LLVM_DIR} (${LLVM_PACKAGE_VERSION}); external library is not instrumented")
endif()
include(cmake/Targets.cmake)

add_subdirectory(toolchain/types)
add_subdirectory(toolchain/target)
add_subdirectory(toolchain/source)
add_subdirectory(toolchain/diagnostic)
add_subdirectory(toolchain/lex)
add_subdirectory(toolchain/parse)
add_subdirectory(toolchain/numeric)
add_subdirectory(toolchain/sem_ir)
add_subdirectory(toolchain/check)
add_subdirectory(toolchain/ir)
add_subdirectory(toolchain/flow)
add_subdirectory(toolchain/lower)
if(NOT CHTHOLLY_FRONTEND_ONLY)
  add_subdirectory(toolchain/runtime)
  add_subdirectory(toolchain/codegen)
  add_subdirectory(toolchain/driver)
endif()
if(BUILD_TESTING)
  include(cmake/ChthollyTesting.cmake)
  add_subdirectory(testing)
endif()
include(cmake/Architecture.cmake)
